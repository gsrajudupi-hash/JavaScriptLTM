/* * Banking System * Demonstrating: * 1. User Defined Functions * 2. Anonymous Functions * 3. Arrow Functions */ // Bank Accounts let accounts = [ { accountNo: 101, name: "Rajesh", balance: 50000 }, { accountNo: 102, name: "Ravi", balance: 35000 }, { accountNo: 103, name: "Priya", balance: 45000 } ]; // ===================================================== // 1. USER DEFINED FUNCTIONS // ===================================================== // Function to display account details function displayAccount(account) { document.write("
"); document.write("Account No : " + account.accountNo + "
"); document.write("Name : " + account.name + "
"); document.write("Balance : ₹" + account.balance + "
"); document.write("
"); } // Function to deposit money function deposit(account, amount) { if (amount > 0) { account.balance = account.balance + amount; document.write( "₹" + amount + " deposited successfully.
" ); document.write( "Current Balance : ₹" + account.balance + "
" ); } else { document.write( "Invalid deposit amount.
" ); } } // Function to withdraw money function withdraw(account, amount) { if (amount <= 0) { document.write( "Invalid withdrawal amount.
" ); } else if (amount > account.balance) { document.write( "Insufficient balance.
" ); } else { account.balance = account.balance - amount; document.write( "₹" + amount + " withdrawn successfully.
" ); document.write( "Current Balance : ₹" + account.balance + "
" ); } } // Function to find an account function findAccount(accountNo) { for (let account of accounts) { if (account.accountNo === accountNo) { return account; } } return null; } // ===================================================== // 2. ANONYMOUS FUNCTIONS // ===================================================== // Anonymous function assigned to a variable const calculateInterest = function (balance, rate) { return balance * rate / 100; }; // Anonymous function to display a message const showMessage = function (message) { document.write( "

" + message + "

" ); }; // ===================================================== // 3. ARROW FUNCTIONS // ===================================================== // Arrow function to calculate balance after interest const calculateBalanceWithInterest = (balance, interest) => balance + interest; // Arrow function to check minimum balance const checkMinimumBalance = balance => balance >= 1000; // Arrow function to get account holder name const getAccountName = account => account.name; // ===================================================== // BANKING SYSTEM // ===================================================== document.write( "

Banking System

" ); document.write( "

Account Details

" ); // Display all accounts accounts.forEach(function (account) { displayAccount(account); }); // ===================================================== // FIND ACCOUNT // ===================================================== document.write( "

Find Account

" ); let account = findAccount(102); if (account != null) { displayAccount(account); } else { document.write( "Account not found.
" ); } // ===================================================== // DEPOSIT // ===================================================== document.write( "

Deposit Operation

" ); deposit(account, 5000); // ===================================================== // WITHDRAW // ===================================================== document.write( "

Withdrawal Operation

" ); withdraw(account, 3000); // ===================================================== // ANONYMOUS FUNCTION // ===================================================== document.write( "

Interest Calculation

" ); let interest = calculateInterest(account.balance, 5); document.write( "Current Balance : ₹" + account.balance + "
" ); document.write( "Interest Rate : 5%
" ); document.write( "Interest : ₹" + interest + "
" ); // ===================================================== // ARROW FUNCTION // ===================================================== document.write( "

Balance After Interest

" ); let newBalance = calculateBalanceWithInterest( account.balance, interest ); document.write( "Balance After Interest : ₹" + newBalance + "
" ); // ===================================================== // ARROW FUNCTION - MINIMUM BALANCE // ===================================================== document.write( "

Minimum Balance Check

" ); document.write( "Minimum Balance Requirement : ₹1000
" ); document.write( "Account Balance : ₹" + account.balance + "
" ); document.write( "Valid Minimum Balance : " + checkMinimumBalance(account.balance) + "
" ); // ===================================================== // ARROW FUNCTION - ACCOUNT NAME // ===================================================== document.write( "

Account Holder

" ); document.write( "Account Holder Name : " + getAccountName(account) + "
" ); // ===================================================== // FILTER USING ARROW FUNCTION // ===================================================== document.write( "

Accounts with Balance Greater Than ₹40,000

" ); let highBalanceAccounts = accounts.filter( account => account.balance > 40000 ); highBalanceAccounts.forEach(account => { document.write( account.accountNo + " - " + account.name + " - ₹" + account.balance + "
" ); }); // ===================================================== // MAP USING ARROW FUNCTION // ===================================================== document.write( "

Account Holder Names

" ); let accountNames = accounts.map( account => account.name ); document.write( accountNames.join(", ") + "
" ); // ===================================================== // FINAL ACCOUNT DETAILS // ===================================================== showMessage("Final Account Details"); accounts.forEach(account => { displayAccount(account); });
